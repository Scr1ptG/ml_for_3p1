import cv2
cap = cv2.VideoCapture(0)            # 0 — веб-камера

while True:
    ok, frame = cap.read()           # 1. прочитать кадр
    if not ok:
        break
    frame = cv2.flip(frame, 1)       # 2. обработать #!
    cv2.imshow("video", frame)       # 3. показать
    if cv2.waitKey(1) == ord("q"):   # нажали q — выходим
        break
2 мин. Этот цикл — скелет и нашего проекта, и контрольной. Меняется только шаг 2. waitKey(1) ждёт 1 мс нажатия клавиши и одновременно даёт окну перерисоваться.
Кнопки и состояние

Нажал клавишу — эффект включился

state = {"caption": False}           # состояние: что включено

def process(frame, key, state):
    if key == ord("c"):              # нажали c #!
        state["caption"] = not state["caption"]   # переключили #!
    if state["caption"]:
        cv2.putText(frame, "IThub Video", (20, 40),
                    cv2.FONT_HERSHEY_SIMPLEX, 1.2,
                    (255, 255, 255), 2)
    return frame

not переворачивает значение:
False → True → False

Это и есть контрольная:
s — цветокоррекция,
t — поворот на 180°

2 мин. Состояние — память программы между кадрами. Без него эффект горел бы только в тот кадр, когда нажата клавиша. Скажите, что контрольная будет ровно на этот шаблон — им понадобится только поменять обработку.

cap = cv2.VideoCapture("vtest.avi")
frames = []
while len(frames) < 80:
    ok, frame = cap.read()          # очередной кадр
    if not ok:
        break
    frames.append(frame)

result = []
for frame in frames:                # меняем каждый кадр #!
    result.append(cv2.flip(frame, 1))
show_video(result)
